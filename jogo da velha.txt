import java.util.Random;
import java.util.Scanner;

public class JogodaVelha {

    public static void main(String[] args) {

        Scanner ler = new Scanner(System.in);
        Random aleatorio = new Random();

        int[][] tabuleiro = new int[3][3];

        int jogadas = 0;
        int fim = 0;

        System.out.println("=== JOGO DA VELHA ===");
        System.out.println("Você é X");
        System.out.println("Computador é O");

        while (fim == 0) {

            mostrarTabuleiro(tabuleiro);

            int linha;
            int coluna;

            do {
                System.out.print("Digite a linha (1-3): ");
                linha = ler.nextInt();

                System.out.print("Digite a coluna (1-3): ");
                coluna = ler.nextInt();

                linha--;
                coluna--;

                if (linha < 0 || linha > 2 ||
                    coluna < 0 || coluna > 2) {

                    System.out.println("Posição inválida!");

                } else if (tabuleiro[linha][coluna] != 0) {

                    System.out.println("Essa posição já foi usada!");
                }

            } while (linha < 0 || linha > 2 ||
                     coluna < 0 || coluna > 2 ||
                     tabuleiro[linha][coluna] != 0);

            tabuleiro[linha][coluna] = 1;
            jogadas++;

            if (verificarVitoria(tabuleiro, 1) == 1) {

                mostrarTabuleiro(tabuleiro);
                System.out.println("Você ganhou!");
                fim = 1;
                break;
            }

            if (jogadas == 9) {

                mostrarTabuleiro(tabuleiro);
                System.out.println("Empate!");
                fim = 1;
                break;
            }

            int linhaComputador;
            int colunaComputador;

            do {

                linhaComputador = aleatorio.nextInt(3);
                colunaComputador = aleatorio.nextInt(3);

            } while (tabuleiro[linhaComputador][colunaComputador] != 0);

            tabuleiro[linhaComputador][colunaComputador] = 2;
            jogadas++;

            System.out.println("O computador jogou!");

            if (verificarVitoria(tabuleiro, 2) == 1) {

                mostrarTabuleiro(tabuleiro);
                System.out.println("O computador ganhou!");
                fim = 1;
                break;
            }

            if (jogadas == 9) {

                mostrarTabuleiro(tabuleiro);
                System.out.println("Empate!");
                fim = 1;
            }
        }

    }


    public static void mostrarTabuleiro(int[][] tabuleiro) {

        System.out.println();

        for (int i = 0; i < 3; i++) {

            for (int j = 0; j < 3; j++) {

                if (tabuleiro[i][j] == 0) {
                    System.out.print(" ");
                }

                if (tabuleiro[i][j] == 1) {
                    System.out.print("X");
                }

                if (tabuleiro[i][j] == 2) {
                    System.out.print("O");
                }

                if (j < 2) {
                    System.out.print(" | ");
                }
            }

            System.out.println();

            if (i < 2) {
                System.out.println("---------");
            }
        }

        System.out.println();
    }


    public static int verificarVitoria(int[][] tabuleiro, int jogador) {

        for (int i = 0; i < 3; i++) {

            if (tabuleiro[i][0] == jogador &&
                tabuleiro[i][1] == jogador &&
                tabuleiro[i][2] == jogador) {

                return 1;
            }
        }

        for (int i = 0; i < 3; i++) {

            if (tabuleiro[0][i] == jogador &&
                tabuleiro[1][i] == jogador &&
                tabuleiro[2][i] == jogador) {

                return 1;
            }
        }

        if (tabuleiro[0][0] == jogador &&
            tabuleiro[1][1] == jogador &&
            tabuleiro[2][2] == jogador) {

            return 1;
        }
        if (tabuleiro[0][2] == jogador &&
            tabuleiro[1][1] == jogador &&
            tabuleiro[2][0] == jogador) {

            return 1;
        }

        return 0;
    }
}
